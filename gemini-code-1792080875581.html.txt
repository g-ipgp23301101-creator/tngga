        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'roll') {
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.15);
            } else if (type === 'ladder') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 0.4);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.4);
            } else if (type === 'snake') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(600, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(200, audioCtx.currentTime + 0.4);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.4);
            } else if (type === 'correct') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
                osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1);
                osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.2);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.4);
            } else if (type === 'wrong') {
                osc.type = 'square';
                osc.frequency.setValueAtTime(250, audioCtx.currentTime);
                osc.frequency.setValueAtTime(180, audioCtx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.3);
            }
        }

        function buildBoard() {
            const gridEl = document.getElementById('boardGrid');
            gridEl.innerHTML = '';

            const matrix = [
                [30, 29, 28, 27, 26, 25],
                [19, 20, 21, 22, 23, 24],
                [18, 17, 16, 15, 14, 13],
                [7, 8, 9, 10, 11, 12],
                [6, 5, 4, 3, 2, 1]
            ];

            for (let r = 0; r < BOARD_ROWS; r++) {
                for (let c = 0; c < BOARD_COLS; c++) {
                    const tileNum = matrix[r][c];
                    const cell = document.createElement('div');
                    cell.className = `cell ${tileNum % 2 === 0 ? 'even' : 'odd'}`;
                    cell.id = `cell-${tileNum}`;

                    if (tileNum === 1) cell.classList.add('start');
                    if (tileNum === 30) cell.classList.add('finish');
                    if (QUESTION_TILES.includes(tileNum)) cell.classList.add('question-tile');

                    let icon = '';
                    if (tileNum === 1) icon = '🚀 MULA';
                    else if (tileNum === 30) icon = '🏁 TAMAT';
                    else if (QUESTION_TILES.includes(tileNum)) icon = '❓';
                    else if (LADDERS[tileNum]) icon = '🪜';
                    else if (SNAKES[tileNum]) icon = '🐍';

                    cell.innerHTML = `
                        <div class="cell-number">${tileNum}</div>
                        <div class="cell-icon">${icon}</div>
                    `;
                    gridEl.appendChild(cell);
                }
            }

            createTokens();
            setTimeout(drawSnakesAndLadders, 100);
        }

        function createTokens() {
            const tokensLayer = document.getElementById('tokensLayer');
            tokensLayer.innerHTML = '';

            GROUPS.forEach(g => {
                const token = document.createElement('div');
                token.className = `token token-${g.id}`;
                token.id = `token-${g.id}`;
                token.innerText = `K${g.id}`;
                tokensLayer.appendChild(token);
            });

            updateTokenPositions();
        }

        function getCellCoordinates(tileNum) {
            const cell = document.getElementById(`cell-${tileNum}`);
            const board = document.querySelector('.board-wrapper');
            if (!cell || !board) return { x: 0, y: 0 };

            const cellRect = cell.getBoundingClientRect();
            const boardRect = board.getBoundingClientRect();

            return {
                x: cellRect.left - boardRect.left + cellRect.width / 2,
                y: cellRect.top - boardRect.top + cellRect.height / 2
            };
        }

        function updateTokenPositions() {
            GROUPS.forEach(g => {
                const token = document.getElementById(`token-${g.id}`);
                if (!token) return;

                const coords = getCellCoordinates(g.pos);
                const offsetX = ((g.id - 1) % 3 - 1) * 10;
                const offsetY = (Math.floor((g.id - 1) / 3) - 0.5) * 10;

                token.style.left = `${coords.x + offsetX}px`;
                token.style.top = `${coords.y + offsetY}px`;
            });
        }

        function drawSnakesAndLadders() {
            const canvas = document.getElementById('snakesLaddersCanvas');
            const boardWrapper = document.querySelector('.board-wrapper');
            if (!canvas || !boardWrapper) return;

            const rect = boardWrapper.getBoundingClientRect();
            canvas.width = rect.width - 20;
            canvas.height = rect.height - 20;

            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            for (let [start, end] of Object.entries(LADDERS)) {
                const p1 = getCellCoordinates(start);
                const p2 = getCellCoordinates(end);

                ctx.strokeStyle = '#22c55e';
                ctx.lineWidth = 8;
                ctx.lineCap = 'round';

                ctx.beginPath();
                ctx.moveTo(p1.x - 10, p1.y - 10);
                ctx.lineTo(p2.x - 10, p2.y - 10);
                ctx.moveTo(p1.x + 10, p1.y + 10);
                ctx.lineTo(p2.x + 10, p2.y + 10);
                ctx.stroke();

                ctx.lineWidth = 4;
                const steps = 5;
                for (let i = 1; i < steps; i++) {
                    const rx1 = p1.x - 10 + (p2.x - p1.x) * (i / steps);
                    const ry1 = p1.y - 10 + (p2.y - p1.y) * (i / steps);
                    const rx2 = p1.x + 10 + (p2.x - p1.x) * (i / steps);
                    const ry2 = p1.y + 10 + (p2.y - p1.y) * (i / steps);
                    ctx.beginPath();
                    ctx.moveTo(rx1, ry1);
                    ctx.lineTo(rx2, ry2);
                    ctx.stroke();
                }
            }

            for (let [head, tail] of Object.entries(SNAKES)) {
                const p1 = getCellCoordinates(head);
                const p2 = getCellCoordinates(tail);

                ctx.strokeStyle = '#ef4444';
                ctx.lineWidth = 10;
                ctx.lineCap = 'round';

                ctx.beginPath();
                ctx.moveTo(p1.x - 10, p1.y - 10);
                const midX = (p1.x + p2.x) / 2 + 30;
                const midY = (p1.y + p2.y) / 2 - 30;
                ctx.quadraticCurveTo(midX, midY, p2.x - 10, p2.y - 10);
                ctx.stroke();

                ctx.fillStyle = '#b91c1c';
                ctx.beginPath();
                ctx.arc(p1.x - 10, p1.y - 10, 10, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function rollDice() {
            if (isRolling) return;

            isRolling = true;
            document.getElementById('rollBtn').disabled = true;

            const diceEl = document.getElementById('diceDisplay');
            diceEl.classList.add('rolling');
            playSound('roll');

            let count = 0;
            const rollInterval = setInterval(() => {
                const tempRoll = Math.floor(Math.random() * 6) + 1;
                diceEl.innerText = tempRoll;
                count++;

                if (count > 10) {
                    clearInterval(rollInterval);
                    diceEl.classList.remove('rolling');
                    const finalRoll = Math.floor(Math.random() * 6) + 1;
                    diceEl.innerText = finalRoll;

                    moveCurrentPlayer(finalRoll);
                }
            }, 50);
        }

        function moveCurrentPlayer(steps) {
            const group = GROUPS[currentGroupIndex];
            let targetPos = group.pos + steps;

            if (targetPos > TOTAL_TILES) {
                targetPos = TOTAL_TILES;
            }

            const moveInterval = setInterval(() => {
                if (group.pos < targetPos) {
                    group.pos++;
                    updateTokenPositions();
                } else {
                    clearInterval(moveInterval);
                    checkTileEffects(group);
                }
            }, 300);
        }

        function checkTileEffects(group) {
            const pos = group.pos;

            if (LADDERS[pos]) {
                setTimeout(() => {
                    playSound('ladder');
                    group.pos = LADDERS[pos];
                    updateTokenPositions();
                    checkPostMove(group);
                }, 400);
            } else if (SNAKES[pos]) {
                setTimeout(() => {
                    playSound('snake');
                    group.pos = SNAKES[pos];
                    updateTokenPositions();
                    checkPostMove(group);
                }, 400);
            } else {
                checkPostMove(group);
            }
        }

        function checkPostMove(group) {
            const pos = group.pos;

            if (QUESTION_TILES.includes(pos)) {
                setTimeout(() => {
                    triggerQuestionModal(group);
                }, 300);
            } else {
                finishTurn();
            }
        }

        function triggerQuestionModal(group) {
            if (questionPool.length === 0) {
                questionPool = [...QUESTIONS];
            }

            const randIdx = Math.floor(Math.random() * questionPool.length);
            activeQuestion = questionPool[randIdx];
            questionPool.splice(randIdx, 1);

            document.getElementById('modalGroupTitle').innerText = `${group.name} — SOALAN ETIKA`;
            document.getElementById('modalGroupTitle').style.color = group.color;
            document.getElementById('modalQuestionText').innerText = activeQuestion.question;

            const optionsContainer = document.getElementById('optionsContainer');
            optionsContainer.innerHTML = '';
            selectedOptionIndex = null;

            activeQuestion.options.forEach((optText, idx) => {
                const optBtn = document.createElement('button');
                optBtn.className = 'option-btn';
                const badgeText = String.fromCharCode(65 + idx);
                optBtn.innerHTML = `<span class="opt-badge">${badgeText}</span> ${optText.substring(3)}`;
                optBtn.onclick = () => selectOption(idx, optBtn);
                optionsContainer.appendChild(optBtn);
            });

            document.getElementById('feedbackBox').className = 'feedback-box';
            document.getElementById('feedbackBox').style.display = 'none';
            document.getElementById('confirmAnsBtn').style.display = 'inline-flex';
            document.getElementById('nextTurnBtn').style.display = 'none';

            document.getElementById('questionModal').classList.add('active');
        }

        function selectOption(index, btnEl) {
            selectedOptionIndex = index;
            document.querySelectorAll('.option-btn').forEach(btn => btn.classList.remove('selected'));
            btnEl.classList.add('selected');
        }

        function confirmAnswer() {
            if (selectedOptionIndex === null) {
                alert("Sila pilih satu jawapan dahulu!");
                return;
            }

            const isCorrect = (selectedOptionIndex === activeQuestion.correct);
            const feedbackBox = document.getElementById('feedbackBox');

            if (isCorrect) {
                playSound('correct');
                feedbackBox.innerText = "JAWAPAN BETUL! 🎉 (+2 Mata untuk kumpulan ini, +1 Mata untuk kumpulan lain)";
                feedbackBox.className = "feedback-box correct";

                GROUPS.forEach((g, idx) => {
                    if (idx === currentGroupIndex) g.score += 2;
                    else g.score += 1;
                });
            } else {
                playSound('wrong');
                const correctOptLetter = String.fromCharCode(65 + activeQuestion.correct);
                feedbackBox.innerText = `JAWAPAN KURANG TEPAT! ❌ Jawapan sebenar ialah ${correctOptLetter}. (+1 Mata untuk kumpulan ini, +2 Mata untuk kumpulan lain)`;
                feedbackBox.className = "feedback-box wrong";

                GROUPS.forEach((g, idx) => {
                    if (idx === currentGroupIndex) g.score += 1;
                    else g.score += 2;
                });
            }

            updateScoreboard();
            document.getElementById('confirmAnsBtn').style.display = 'none';
            document.getElementById('nextTurnBtn').style.display = 'inline-flex';
        }

        function closeQuestionModal() {
            document.getElementById('questionModal').classList.remove('active');
            finishTurn();
        }

        function finishTurn() {
            if (GROUPS[currentGroupIndex].pos >= TOTAL_TILES) {
                endGame("Kumpulan telah mencapai Petak TAMAT!");
                return;
            }

            currentGroupIndex = (currentGroupIndex + 1) % GROUPS.length;
            updateUIState();
            isRolling = false;
            document.getElementById('rollBtn').disabled = false;
        }

        function updateUIState() {
            const currentGroup = GROUPS[currentGroupIndex];
            const turnDisplay = document.getElementById('turnDisplay');
            turnDisplay.innerText = `Giliran: ${currentGroup.name}`;
            turnDisplay.style.backgroundColor = currentGroup.color;

            updateScoreboard();
        }

        function updateScoreboard() {
            const grid = document.getElementById('scoresGrid');
            grid.innerHTML = '';

            GROUPS.forEach((g, idx) => {
                const item = document.createElement('div');
                item.className = `score-item ${idx === currentGroupIndex ? 'active-group' : ''}`;
                item.innerHTML = `
                    <div class="score-tag">
                        <div class="color-dot" style="background: ${g.color};"></div>
                        <span>${g.name}</span>
                    </div>
                    <div class="score-value">${g.score}</div>
                `;
                grid.appendChild(item);
            });
        }

        function startTimer() {
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                timerSeconds--;
                updateTimerDisplay();

                if (timerSeconds <= 0) {
                    clearInterval(timerInterval);
                    endGame("Masa 10 minit telah tamat!");
                }
            }, 1000);
        }

        function updateTimerDisplay() {
            const mins = Math.floor(timerSeconds / 60);
            const secs = timerSeconds % 60;
            document.getElementById('timerDisplay').innerText = `⏳ ${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
        }

        function endGame(reason) {
            clearInterval(timerInterval);
            document.getElementById('rollBtn').disabled = true;

            const sorted = [...GROUPS].sort((a, b) => b.score - a.score);
            const maxScore = sorted[0].score;
            const winners = sorted.filter(g => g.score === maxScore);

            const winnerBox = document.getElementById('winnerAnnouncement');
            if (winners.length === 1) {
                winnerBox.innerText = `🏆 PEMENANG: ${winners[0].name} dengan ${winners[0].score} Mata!`;
            } else {
                const winnerNames = winners.map(w => w.name).join(' & ');
                winnerBox.innerText = `🤝 KEPUTUSAN SERI: ${winnerNames} dengan ${maxScore} Mata!`;
            }

            const listEl = document.getElementById('finalScoresList');
            listEl.innerHTML = '';
            sorted.forEach((g, idx) => {
                const row = document.createElement('div');
                row.style.display = 'flex';
                row.style.justifyContent = 'space-between';
                row.style.padding = '8px 15px';
                row.style.borderRadius = '8px';
                row.style.background = '#f1f5f9';
                row.style.fontWeight = 'bold';
                row.innerHTML = `<span>#${idx + 1} ${g.name}</span> <span>${g.score} Mata (Petak ${g.pos})</span>`;
                listEl.appendChild(row);
            });

            document.getElementById('gameOverModal').classList.add('active');
        }

        function closeGameOverModal() {
            document.getElementById('gameOverModal').classList.remove('active');
        }

        function openRulesModal() {
            document.getElementById('rulesModal').classList.add('active');
        }

        function closeRulesModal() {
            document.getElementById('rulesModal').classList.remove('active');
        }

        function openScoreModal() {
            updateScoreboard();
            alert("Markah dikemas kini pada panel sisi!");
        }

        function confirmReset() {
            if (confirm("Adakah anda pasti mahu memulakan semula permainan? Markah dan kedudukan akan ditetapkan semula.")) {
                resetGame();
            }
        }

        function resetGame() {
            clearInterval(timerInterval);
            timerSeconds = 600;
            currentGroupIndex = 0;
            isRolling = false;
            questionPool = [...QUESTIONS];

            GROUPS.forEach(g => {
                g.pos = 1;
                g.score = 0;
            });

            document.getElementById('rollBtn').disabled = false;
            document.getElementById('diceDisplay').innerText = '🎲';

            updateTokenPositions();
            updateUIState();
            updateTimerDisplay();
            startTimer();
        }

        window.addEventListener('resize', () => {
            drawSnakesAndLadders();
            updateTokenPositions();
        });

        window.onload = () => {
            buildBoard();
            updateUIState();
            startTimer();
        };
    </script>
